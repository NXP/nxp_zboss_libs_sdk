◆ zb_zdo_active_scan_request()

void zb_zdo_active_scan_request ( zb_uint8_t  param)

Performs active scan.

The result will be provided via callback that is passed within zb_zdo_active_scan_request_t. The only argument from this callback is the index of a buffer with zb_zdo_active_scan_resp_t param, followed by a sequence of zb_nlme_network_descriptor_t params (count is determined by zb_zdo_active_scan_resp_t - network_count)

Sample use of active scan request:

static void active_scan_complete_cb(zb_uint8_t param)
{
zb_zdo_active_scan_resp_t *cnf;
zb_nlme_network_descriptor_t *nwk_descr;
zb_ext_pan_id_t extended_pan_id;
zb_uint16_t short_pan_id;
cnf = (zb_zdo_active_scan_resp_t *)zb_buf_begin(param);
TRACE_MSG(TRACE_APP1, "Scan results: status=%d, network_count=%hd", (FMT__D_H, cnf->status, cnf->network_count));
if (cnf->status == RET_OK)
{
nwk_descr = (zb_nlme_network_descriptor_t *)(cnf + 1);
for (zb_uint32_t i = 0; i < cnf->network_count; i++)
{
zb_address_get_pan_id(nwk_descr->panid_ref, extended_pan_id);
zb_address_get_short_pan_id(nwk_descr->panid_ref, &short_pan_id);
TRACE_MSG(TRACE_APP1, "NWK #%d: short PAN ID=0x%x, extended PAN ID=" TRACE_FORMAT_64 " , permit_joining=%d, router_capacity=%d, end_device_capacity=%d, nwk_update_id=%hd",
(FMT__D_H_A_D_D_D_H,
i,
short_pan_id,
TRACE_ARG_64(extended_pan_id),
nwk_descr->permit_joining,
nwk_descr->router_capacity,
nwk_descr->end_device_capacity,
nwk_descr->nwk_update_id
));
nwk_descr++;
}
}
}
{
zb_bufid_t buf;
zb_zdo_active_scan_request_t *req;
buf = zb_buf_get_out();
if (buf == 0)
{
TRACE_MSG(TRACE_ERROR, "Can't allocate buffer!", (FMT__0));
break;
}
req = ZB_BUF_GET_PARAM(buf, zb_zdo_active_scan_request_t);
req->cb = active_scan_complete_cb;
req->scan_channels_list[0] = (1L << CHANNEL);
req->scan_duration = ZB_DEFAULT_SCAN_DURATION;
ZB_SCHEDULE_APP_CALLBACK(zb_zdo_active_scan_request, buf);
}
Parameters
param- index of buffer with zb_nlme_network_discovery_request_t param

◆ zb_zdo_ed_scan_request()

void zb_zdo_ed_scan_request ( zb_uint8_t  param)

Performs energy scan.

The result of scan will be provided via callback that is passed within zb_zdo_ed_scan_param_t. The result buffer contains zb_energy_detect_list_t structure in its tail and can be acquired using ZB_BUF_GET_PARAM macro. The status of scan passed via buffer status and can be acquired using zb_buf_get_status macro.

Sample usage of energy scan:

zb_zdo_ed_scan_param_t *scan_params = ZB_BUF_GET_PARAM(param, zb_zdo_ed_scan_param_t);
ZB_BZERO(scan_params, sizeof(*scan_params));
TRACE_MSG(TRACE_APP1, "perform_ed_scan %hd", (FMT__H, param));
scan_params->cb = ed_scan_completed_cb;
scan_params->nwk_param.scan_duration = ED_SCAN_DURATION;
scan_params->nwk_param.scan_channels_list[0] = CHANNEL_MASK;
ZB_SCHEDULE_APP_CALLBACK(zb_zdo_ed_scan_request, param);
static void ed_scan_completed_cb(zb_bufid_t param)
{
static zb_bool_t device_started = ZB_FALSE;
zb_energy_detect_list_t *ed_list = ZB_BUF_GET_PARAM(param, zb_energy_detect_list_t);
zb_uint_t result_idx;
zb_uint_t min_energy_idx = 0;
TRACE_MSG(TRACE_APP1, "ed_scan_completed_cb param %hd, status %d, channel_count %hd",
(FMT__H_D_H, param, zb_buf_get_status(param), ed_list->channel_count));
if (zb_buf_get_status(param) == RET_OK)
{
for (result_idx = 0; result_idx < ed_list->channel_count; result_idx++)
{
TRACE_MSG(TRACE_APP1, "result idx: %d, channel_number: %hd, page_idx: %hd, energy_detected: %hd",
(FMT__D_H_H_H, result_idx,
ed_list->channel_info[result_idx].channel_number,
ed_list->channel_info[result_idx].channel_page_idx,
ed_list->channel_info[result_idx].energy_detected));
if (ed_list->channel_info[result_idx].energy_detected <= ed_list->channel_info[min_energy_idx].energy_detected)
{
min_energy_idx = result_idx;
}
}
if (!device_started)
{
zb_set_channel_mask(1 << ed_list->channel_info[min_energy_idx].channel_number);
}
}
if (!device_started)
{
device_started = ZB_TRUE;
zboss_start_continue();
}
zb_buf_free(param);
}
Parameters
param- index of buffer that has zb_zdo_ed_scan_param_t structure in its tail.
Note
This function can't be used in NCP-host build.

◆ zb_zdo_mgmt_lqi_req()

zb_uint8_t zb_zdo_mgmt_lqi_req ( zb_uint8_t  param,
zb_callback_t  cb 
)

Sends Mgmt_Lqi_req (see Zigbee spec 2.4.3.3.2)

Parameters
param- index of buffer with Lqi request parameters. zb_zdo_mgmt_lqi_param_s
cb- user's function to call when got response from the remote.
Returns
- ZDP transaction sequence number or 0xFF if operation cannot be performed now (nor enough memory, resources, etc.) zb_zdo_mgmt_lqi_resp_s, zb_zdo_neighbor_table_record_s

Example:

{
zb_uint8_t tsn;
zb_zdo_mgmt_lqi_param_t *req_param;
