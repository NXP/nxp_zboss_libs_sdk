Developing with ZBOSS for Zigbee
ZVD Developer's guide

Requirements

  • Ubuntu 20.04 (or 22.04). Other Linux x64 systems are allowed but not tested.
  • PC bluetooth adapter with Bluetooth 4.0 and BLE support.

Environment

Install the following libraries:

  • libglib2.0-dev
  • libgio2.0-dev
  • libbsd-dev
  • libm

Setup Bluez 5.63

Uninstall current bluez version if it is already installed.

Install bluez-5.63:

$: sudo apt install build-essential libdbus-1-dev libglib2.0-dev libical-dev libreadline-dev libudev-dev python3-docutils
$: wget http://www.kernel.org/pub/linux/bluetooth/bluez-5.63.tar.xz
$: tar -xf bluez-5.63.tar.xz
$: cd bluez-5.63
$: ./configure
$: make
$: sudo make install

Check that bluetoothctl --version command prints bluetoothctl: 5.63 string.

Create a ZVD application

Initialization

  • Create MAIN() function
  • Init device as ZVD:
    zb_ret_t ret;
    ARGV_UNUSED;
    ZB_SET_TRACE_ON();
    ZB_SET_TRAF_DUMP_ON();
    ZB_INIT("zvd_central");
    ret = zbd_zvd_init(DEVICE_NAME);
    TRACE_MSG(TRACE_APP1, "ZVD is started, ret %d", (FMT__D, ret));
  • Generate and set EUI-64 address:
    {
    zb_uint8_t ble_address[6];
    zb_ieee_addr_t ieee_addr;
    ret = zbd_ble_addr_get(ble_address);
    ZB_ASSERT(ret == RET_OK);
    TRACE_MSG(TRACE_APP1, "Main: Generated IEEE address: %02X:%02X:%02X:%02X:%02X:%02X",
    (FMT__H_H_H_H_H_H, ble_address[0], ble_address[1], ble_address[2], ble_address[3],
    ble_address[4], ble_address[5]));
    ret = zbd_generate_eui64_from_mac_address(ble_address, ieee_addr);
    ZB_ASSERT(ret == RET_OK);
    TRACE_MSG(TRACE_APP1, "Main: Generated IEEE address: " TRACE_FORMAT_64,
    (FMT__A, TRACE_ARG_64(ieee_addr)));
    zb_set_long_address(ieee_addr);
    }
  • Register ZB ZVD callbacks:
    zbd_zvd_set_zdd_list_updated_callback(zvd_commissioner_periph_list_updated_callback);
    zbd_zvd_set_connect_callback(zvd_commissioner_connect_callback);
    zbd_zvd_set_services_resolved_callback(zvd_commissioner_services_resolved_callback);
    zbd_zvd_set_disconnect_callback(zvd_commissioner_disconnect_callback);
    zbd_zvd_security_set_secure_session_established_callback(
    zvd_commissioner_security_establish_secure_session_callback);
    zbd_zvd_comm_set_status_update_callback(zvd_commissioner_comm_status_update_callback);
    zbd_zvd_set_operation_return_callback(zvd_commissioner_operation_return_callback);

Start ZBOSS

Implement an interface for calling ZVD commands and start ZBOSS.

For example, DSR ZVD Demo app uses a CLI interface

if (zboss_start_no_autostart() == RET_OK)
{
pthread_create(&zvd.cli.thread_id, NULL, &cli_thread, NULL);
while (!zvd.cli.kill_thread && !ZB_SCHEDULER_IS_STOP())
{
zboss_main_loop_iteration();
}
}
else
{
TRACE_MSG(TRACE_ERROR, "Main: can't start ZBOSS", (FMT__0));
}

Connect to ZDD and create centralized network via ZVD

  • Start discovering ZDD devices in range, zbd_zvd_discover_zdd()
    /**
    * @brief Starts discovering ZDD devices in range.
    *
    * @return zb_ret_t
    * - RET_OK on success
    * - RET_ERROR on error
    */
    static zb_ret_t zvd_commissioner_discover_zdd(void)
    {
    zb_ret_t ret;
    if (zvd.connected)
    {
    /* Disconnect before calling discover again */
    return RET_ERROR;
    }
    ret = zbd_zvd_discover_zdd();
    if (ret == RET_OK)
    {
    zvd.wait_for_discovery = ZB_TRUE;
    ZB_SCHEDULE_APP_ALARM(
    discovery_timeout, 0, ZB_MILLISECONDS_TO_BEACON_INTERVAL(ZDD_DISCOVERY_TIMEOUT_ms));
    }
    return ret;
    }
    See also
    zbd_zvd_set_zdd_list_updated_callback()

  • Wait zdd_list_updated_callback and call zbd_zvd_stop_discovery()

    For example:

    /**
    * @brief Callback triggered when there is an update on the ZDD peripherals in range list.
    *
    * @note This function should call @ref zbd_zvd_get_discovered_peripherals_list to get the
    * updated list
    */
    static void zvd_commissioner_periph_list_updated_callback(void)
    {
    /* Check ZVD conditions when receiving this callback */
    if (zvd.connected || !zvd.wait_for_discovery)
    {
    /* If ZVD is already connected or is not waiting for discovery, ignore this callback */
    return;
    }
    zvd.wait_for_discovery = ZB_FALSE;
    /* Cancel discovery timeout alarm */
    ZB_SCHEDULE_APP_ALARM_CANCEL(discovery_timeout, ZB_ALARM_ALL_CB);
    /* Stop discovery */
    zvd_commissioner_stop_discovery();
    /* Clear list of discovered ZDDs */
    ZB_BZERO(zvd.discovered_zdds, sizeof(zvd.discovered_zdds));
    zvd.discovered_zdds_count = 0;
    /* Get the updated list of ZDDs in range */
    zvd.discovered_zdds_count = zbd_zvd_get_discovered_peripherals_list(
    zvd.discovered_zdds, sizeof(zvd.discovered_zdds));
    ZVD_CLI_PRINTLN("Found %u devices", zvd.discovered_zdds_count);
    for (unsigned i = 0; i < zvd.discovered_zdds_count; ++i)
    {
    V_TRACE_MSG(
    "zvd_commissioner_periph_list_updated_callback: found zdd (%02X:%02X:%02X:%02X:%02X:%02X)",
    zvd.discovered_zdds[i].mac_addr[0], zvd.discovered_zdds[i].mac_addr[1],
    zvd.discovered_zdds[i].mac_addr[2], zvd.discovered_zdds[i].mac_addr[3],
    zvd.discovered_zdds[i].mac_addr[4], zvd.discovered_zdds[i].mac_addr[5]);
    ZVD_CLI_PRINTLN("[%d]:", i);
    print_zdd(&zvd.discovered_zdds[i]);
    }
    /* Asynchronous command executed successfully */
    if (zvd.command_in_progress)
    {
    zvd.command_success = ZB_TRUE;
    }
    }
    See also
    zbd_zvd_set_zdd_list_updated_callback()
  • Connect to ZDD, zbd_zvd_connect()
    /**
    * @brief Connects to a designated ZDD in range using its BLE address
    *
    * @param ble_address Pointer to 6-bytes array of the ZDD BLE address
    * @return zb_ret_t
    * - RET_OK on success
    * - RET_ERROR on error
    */
    static zb_ret_t zvd_commissioner_connect(const zb_uint8_t *ble_address)
    {
    zb_ret_t ret;
    if (zvd.connected)
    {
    return RET_ERROR;
    }
    ret = zbd_zvd_connect(ble_address);
    if (ret == RET_OK)
    {
    zvd.wait_for_connect = ZB_TRUE;
    ZB_SCHEDULE_APP_ALARM(
    connect_timeout_callback, 0, ZB_MILLISECONDS_TO_BEACON_INTERVAL(CONNECT_TIMEOUT_ms));
    }
    return ret;
    }
    See also
    zbd_zvd_set_connect_callback()
  • Form centralized network on ZDD:
    • fill parameter structure ZBD_ZVD_COMM_SERVICE_PARAMS_T
    • call zbd_zvd_comm_form_network()
  • Open network on ZDD, zbd_zvd_comm_permit_joining()
    ret = zbd_zvd_comm_permit_joining(zvd.conn_id, permit_duration);
    if (ret == RET_OK)
    {
    zvd.wait_for_comm_status = ZB_TRUE;
    /* Schedule alarm for commissioning status update timeout */
    ZB_SCHEDULE_APP_ALARM(
    comm_status_update_timeout_callback, 0, ZB_MILLISECONDS_TO_BEACON_INTERVAL(COMM_STATUS_UPDATE_TIMEOUT_ms));
    }
    See also
    zbd_zvd_comm_set_status_update_callback()
