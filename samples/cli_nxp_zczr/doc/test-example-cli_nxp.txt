endpoint create 10 HA MT 0
cluster create 10 GB server
cluster create 10 GPLC server
config ieee_addr 00:00:00:00:00:00:00:01
config role end_device
config channel 21
config sleepy
nvram erase_at_start 1
config start
sleep 2
network join centralized
sleep 4

# ZC Binds Poll Control cluster to ZED: Trigger periodic check-in
cluster bind 5 00:00:00:00:00:00:00:01 0x<SED_short_addr> 10 GPLC ieee

# ZC Modifies ZED's check-in interval to 480 quarter-seconds (120 seconds)
cluster write_attr 5 0x<SED_short_addr> 10 GPLC 0x0000 0x23 E0:01:00:00
sleep 2
cluster read_attr 5 0x<SED_short_addr> 10 GPLC 0x0000

# ZC client sets short poll interval on ZED server to 4 quarter seconds (1 second)
cluster pollctrl_cmd set_short_poll_interval 5 0x<SED_short_addr> 10 1
sleep 2

# ZC client sets long poll interval on ZED server to 60 quarter-seconds (15 seconds)
cluster pollctrl_cmd set_long_poll_interval 5 0x<SED_short_addr> 10 60

# ZC sets fast poll timeout to 0x0028 quarter seconds (10 seconds)
cluster write_local 5 GPLC client 0x0003 0x21 28:00

# ZC client stops fast polling on ZED server
cluster pollctrl_cmd fast_poll_stop 5 0x<SED_short_addr> 10
sleep 2

SEQUENCE FOR DISCOVER/READ/WRITE attribute:
===========================================

# On real world we should have:
# - device 1 with cluster ON/OFF server
# - device 2 with cluster ON/OFF Switch server
# here to simplify the test sequence, we configure:
# - all servers on device 1
#-  all test commands on device 2

# ZC as ON/OFF & Switch Server:
endpoint create 5 HA MT 0
cluster create 5 GB server
cluster create 5 GI server
cluster create 5 GOF server
cluster create 5 GOFSC server
cluster create 5 MTM server
config ieee_addr aa:aa:aa:aa:aa:aa:aa:aa
config role coordinator
config panid 0xcafe
config channel 21
config nwk_key 0 ab:cd:ef:01:23:45:67:89:00:00:00:00:00:00:00:00
nvram erase_at_start 1
config start
sleep 2
# This local cluster DOES exist
cluster read_local 5 GOFSC server 0x0010
# This local cluster does NOT exist
cluster read_local 5 GOFSC client 0x0010
# read ON OFF Switch Configuration: SwitchType: enum8, read only
cluster read_local 5 GOFSC server 0x0000
# read ON OFF Switch Configuration: SwitchActions: enum8, read write, range 0-2
cluster read_local 5 GOFSC server 0x0010
# write ON OFF Switch Configuration: SwitchType: enum8, read only
cluster write_local 5 GOFSC server 0x0000 0x30 00
# => result: Error: Read only
# write ON OFF Switch Configuration: SwitchActions: enum8, read write, range 0-2
cluster write_local 5 GOFSC server 0x0010 0x10 01
# => result: Error: Invalid type
cluster write_local 5 GOFSC server 0x0010 0x30 03
# => result: Error: Invalid value
cluster write_local 5 GOFSC server 0x0010 0x30 02
# => result: OK
cluster read_local 5 GOFSC server 0x0010
# => read back value
sleep 2
network form centralized
sleep 2
network open 180
